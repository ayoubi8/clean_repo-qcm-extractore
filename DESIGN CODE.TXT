        function switchView(viewId) {
            // Hide all views
            const views = ['dashboard', 'projects', 'settings', 'project_pipeline', 'correction_studio'];
            views.forEach(v => {
                const el = document.getElementById('view-' + v);
                if (el) el.classList.add('hidden');
                if (el) el.classList.remove('active-view');
            });

            // Reset navigation styles
            const navItems = ['dashboard', 'projects', 'settings'];
            navItems.forEach(item => {
                const el = document.getElementById('nav-' + item);
                if (el) {
                    el.classList.remove('text-cyan-400', 'bg-[#353534]/50', 'border-r-2', 'border-cyan-400');
                    el.classList.add('text-[#869397]', 'hover:text-white', 'hover:bg-[#201f1f]');
                }
            });

            // Show active view
            const activeEl = document.getElementById('view-' + viewId);
            if (activeEl) {
                activeEl.classList.remove('hidden');
                activeEl.classList.add('active-view');
            }

            // Apply active styles to nav if it's a top-level item
            if (navItems.includes(viewId)) {
                const activeNav = document.getElementById('nav-' + viewId);
                if (activeNav) {
                    activeNav.classList.add('text-cyan-400', 'bg-[#353534]/50', 'border-r-2', 'border-cyan-400');
                    activeNav.classList.remove('text-[#869397]', 'hover:text-white', 'hover:bg-[#201f1f]');
                }
            }
            
            // Scroll to top
            document.querySelector('main').scrollTop = 0;
        }

        function switchView(viewId) {
            // Hide all views
            document.querySelectorAll('.view-section').forEach(view => {
                view.classList.remove('active');
            });
            // Show active view
            document.getElementById(viewId).classList.add('active');

            // Scroll to top
            document.getElementById('viewport').scrollTop = 0;

            // Navigation style updates
            document.querySelectorAll('aside button').forEach(btn => {
                btn.classList.remove('text-[#4cd7f6]', 'bg-[#353534]', 'font-bold', 'border-r-2', 'border-[#4cd7f6]');
                btn.classList.add('text-[#869397]');
            });
            
            const activeNavId = viewId.replace('view-', 'nav-');
            const activeBtn = document.getElementById(activeNavId);
            if (activeBtn) {
                activeBtn.classList.remove('text-[#869397]');
                activeBtn.classList.add('text-[#4cd7f6]', 'bg-[#353534]', 'font-bold', 'border-r-2', 'border-[#4cd7f6]');
            }

            // FAB suppression logic
            const fab = document.getElementById('global-fab');
            if (viewId === 'view-dashboard' || viewId === 'view-projects') {
                fab.style.display = 'flex';
            } else {
                fab.style.display = 'none';
            }
        }

        // Terminal Log simulation
        const logsContainer = document.getElementById('terminal-logs');
        const simulatedLogs = [
            "Validating schema mapping for Record #2091...",

            "New medical PDF detected in watch directory /clinical/ingest/oncology",
            "Applying OCR refinement filter: Bilateral Blur..."
        ];

        setInterval(() => {
            if (logsContainer) {
                const log = document.createElement('p');
                const time = new Array(8).fill(0).map(() => Math.floor(Math.random() * 10)).join('');

                logsContainer.appendChild(log);
                logsContainer.scrollTop = logsContainer.scrollHeight;
                if (logsContainer.children.length > 20) logsContainer.removeChild(logsContainer.firstChild);
            }
        }, 3000);

        // Run Step Simulation
        const runBtn = document.getElementById('run-step-btn');
        if (runBtn) {
            runBtn.addEventListener('click', () => {

                runBtn.classList.add('brightness-75', 'cursor-not-allowed');

                        runBtn.innerHTML = '<span class="material-symbols-outlined text-sm">play_arrow</span> Run Step';

                        runBtn.classList.remove('bg-green-500', 'brightness-75', 'cursor-not-allowed');
                    }, 2000);
                }, 3000);
            });
